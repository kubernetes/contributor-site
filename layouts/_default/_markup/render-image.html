{{- /* Based on Hugo's embedded image render hook (v0.144.2), which resolves page bundle and global resources. Defining this file replaces that hook, so its logic is kept here for every other page. Pages from data/remote_docs.yaml carry params.remote. Their relative image paths point at files in the upstream repo, so they load from there instead. */ -}} {{- $u := urls.Parse .Destination -}} {{- $src := $u.String -}} {{- if not $u.IsAbs -}} {{- with .Page.Params.remote -}} {{- if $u.Path -}} {{- $file := path.Join (path.Dir .path) $u.Path -}} {{- if hasPrefix $u.Path "/" -}} {{- $file = strings.TrimPrefix "/" $u.Path -}} {{- end -}} {{- $src = printf "https://raw.githubusercontent.com/%s/%s/%s" .repo .ref $file -}} {{- end -}} {{- else -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- with or ($.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- $src = .RelPermalink -}} {{- with $u.RawQuery -}} {{- $src = printf "%s?%s" $src . -}} {{- end -}} {{- with $u.Fragment -}} {{- $src = printf "%s#%s" $src . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- /**/ -}}
